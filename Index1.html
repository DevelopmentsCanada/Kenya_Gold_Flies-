Kenya Flies - Client Fly Order Portal HTML

Overview & Setup Guide for Patrick

This single standalone HTML application allows clients to browse Patrick's master fly catalog, select quantities in dozens, enter contact details and shipping destinations, calculate order totals automatically, and submit completed orders directly back to Patrick via email or instant text message.

How Patrick Can Share This Tool:

  1. Save the HTML file as index.html or order.html.
  2. Host it on Netlify, GitHub Pages, or send the file directly to clients.
  3. Clients open the link in any browser, fill out their order and shipping address, and click "Submit Order to Patrick via Email" or "Copy Order Summary for WhatsApp".

                            <input type="number" step="0.5" min="0" value="${doz}" onchange="updateQty(${item.id}, this.value)" class="w-16 text-xs bg-slate-50 border border-slate-200 rounded-lg px-2 py-1 font-bold text-center mono focus:bg-white focus:outline-none">

            });

            updateTotals();

        }

        function updateQty(id, val) {

            const num = parseFloat(val) || 0;

            if (num > 0) {

                selectedQuantities[id] = num;

            } else {

                delete selectedQuantities[id];

            }

            updateTotals();

        }

        function updateTotals() {

            let totalDozens = 0;

            let totalCost = 0;

            let patternCount = 0;

            Object.keys(selectedQuantities).forEach(id => {

                const item = MASTER_CATALOG.find(f => f.id == id);

                if (item) {

                    const doz = selectedQuantities[id];

                    totalDozens += doz;

                    totalCost += (doz * item.price);

                    patternCount++;

                }

            });

            const totalFlies = Math.round(totalDozens * 12);

            document.getElementById('metric-dozens').innerText = `${totalDozens.toFixed(1)} doz`;

            document.getElementById('metric-flies').innerText = `${totalFlies} flies`;

            document.getElementById('metric-total').innerText = `$${totalCost.toFixed(2)}`;

            document.getElementById('summary-patterns-count').innerText = patternCount;

            document.getElementById('summary-dozens').innerText = `${totalDozens.toFixed(1)} doz`;

            document.getElementById('summary-flies').innerText = `${totalFlies} flies`;

            document.getElementById('summary-grand-total').innerText = `$${totalCost.toFixed(2)}`;

        }

        function buildOrderManifest() {

            const name = document.getElementById('client-name').value.trim();

            const email = document.getElementById('client-email').value.trim();

            const phone = document.getElementById('client-phone').value.trim();

            const address = document.getElementById('client-address').value.trim();

            const notes = document.getElementById('client-notes').value.trim();

            let manifest = `COMMERCIAL FLY ORDER FOR PATRICK (KENYA FLIES)\n`;

            manifest += `==========================================\n`;

            manifest += `Client Name: ${name || 'N/A'}\n`;

            manifest += `Email: ${email || 'N/A'}\n`;

            manifest += `Phone/WhatsApp: ${phone || 'N/A'}\n`;

            manifest += `Shipping Address: ${address || 'N/A'}\n`;

            if (notes) manifest += `Custom Notes: ${notes}\n`;

            manifest += `==========================================\n\n`;

            manifest += `ITEMIZED FLY ORDER LIST:\n`;

            let totalDoz = 0;

            let grandTotal = 0;

            Object.keys(selectedQuantities).forEach(id => {

                const item = MASTER_CATALOG.find(f => f.id == id);

                if (item) {

                    const doz = selectedQuantities[id];

                    const cost = doz * item.price;

                    totalDoz += doz;

                    grandTotal += cost;

                    manifest += `• ${item.id}. ${item.name} (${item.category})\n`;

                    manifest += `  Qty: ${doz} doz (${Math.round(doz * 12)} flies) | Hook: ${item.hook} | Weight: ${item.weight}\n`;

                    manifest += `  Subtotal: $${cost.toFixed(2)}\n\n`;

                }

            });

            manifest += `==========================================\n`;

            manifest += `TOTAL DOZENS: ${totalDoz.toFixed(1)} doz (${Math.round(totalDoz * 12)} flies)\n`;

            manifest += `GRAND TOTAL: $${grandTotal.toFixed(2)}\n`;

            return { name, email, phone, address, notes, manifest, grandTotal, totalDoz };

        }

        function sendOrderEmail() {

            const data = buildOrderManifest();

            if (!data.name || !data.email || !data.address) {

                alert('Please fill in your Name, Email, and Shipping Address before submitting your order.');

                return;

            }

            if (data.totalDoz === 0) {

                alert('Please select at least 0.5 dozen of a fly pattern to place an order.');

                return;

            }

            const subject = encodeURIComponent(`New Fly Order from ${data.name} - ${data.totalDoz.toFixed(1)} Dozen ($${data.grandTotal.toFixed(2)})`);

            const body = encodeURIComponent(data.manifest);

           

            // Default recipient: Patrick

            window.location.href = `mailto:patrick@kenyaflies.com?subject=${subject}&body=${body}`;

        }
